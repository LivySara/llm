<template>
    <bk-dialog
        v-model="visible"
        theme="primary"
        width="640"
        :mask-close="false"
        :auto-close="false"
        :loading="loading"
        header-position="left"
        :title="dialogTitle"
        @confirm="handleConfirm"
        @cancel="handleCancel">
        <bk-alert
            class="approve-tip"
            type="warning"
            :closable="false"
            :title="tipText" />
    </bk-dialog>
</template>

<script>
import { defHttp, Message, bkInfo, $t, loginInfo } from 'platform'

export default {
    name: 'custom-approve-button',
    inject: {
        refreshTicket: {
            default: null
        }
    },
    props: {
        form: {
            type: Object,
            default: () => ({})
        },
        button: {
            type: Object,
            default: () => ({})
        }
    },
    data() {
        return {
            visible: false,
            loading: false,
            dialogTitle: $t('审批同意'),
            tipText: ''
        }
    },
    methods: {
        getTipText() {
            const field = (this.form.fields || []).find(item => item.key === 'tip_value')
            if (field) {
                // 隐藏字段会被 getFormData() 过滤，直接取字段原始值
                if (field.state === 'hide') {
                    return field.value ?? ''
                }
                const data = this.form.getFormData?.() || {}
                return data.tip_value ?? field.value ?? ''
            }
            const data = this.form.getFormData?.() || {}
            return data.tip_value ?? ''
        },
        getApproveButtonMeta() {
            const activity = this.form.activityInfo || {}
            const code = this.button?.button_code || this.button?.key || 'approve'
            const runtimeBtn = (activity.buttons || []).find(item => (item.button_code || item.key) === code) || {}
            const metaBtn = (activity.activity_meta?.buttons || []).find(item => (item.button_code || item.key) === code) || {}
            const button = { ...metaBtn, ...runtimeBtn }
            return {
                code,
                button,
                meta: { ...(metaBtn.meta || {}), ...(runtimeBtn.meta || {}) }
            }
        },
        openDialog() {
            console.log('调式');
            const ticketInfo = this.form.ticketInfo || {}
            const activityInfo = this.form.activityInfo || {}
            if (!ticketInfo.id || !activityInfo.id) {
                Message.error($t('当前节点没有可审批的任务'))
                return
            }
            const { button } = this.getApproveButtonMeta()
            this.dialogTitle = button.name ? $t('审批{0}', [button.name]) : $t('审批同意')
            this.tipText = this.getTipText()
            this.visible = true
        },
        async handleClick() {
            const check = await this.form.formCheck()

            if (check && check.code === 'error') {
                Message.error(check.message || $t('表单校验失败'))
                return
            }
            if (check && check.code === 'warning') {
                bkInfo({
                    type: 'warning',
                    theme: 'primary',
                    title: $t('确认提交'),
                    subTitle: check.message,
                    cancelText: $t('确定'),
                    crossAsCancel: false,
                    okText: $t('返回检查'),
                    confirmFn: () => {},
                    cancelFn: (val, cancelType) => {
                        cancelType === 'cancel' && this.openDialog()
                    }
                })
                return
            }

            this.openDialog()
        },
        handleCancel() {
            this.visible = false
        },
        async handleConfirm() {
            this.loading = true
            try {
                const { code, button } = this.getApproveButtonMeta()
                const ticketInfo = this.form.ticketInfo || {}
                const activityInfo = this.form.activityInfo || {}
                const operateBtnData = {
                    view: this.tipText,
                    method: 'approve',
                    button_code: code
                }

                if (typeof this.button?.runAction === 'function') {
                    await this.button.runAction(operateBtnData)
                }

                this.dialogTitle = button.name ? $t('审批{0}', [button.name]) : $t('审批同意')

                const params = {
                    id: ticketInfo.id,
                    task_id: activityInfo.id,
                    operator: {
                        type: 'user',
                        id: loginInfo && loginInfo.username
                    },
                    form_data: {
                      ...this.form.getFormData(),
                      tip_value: this.tipText
                    },
                    options: this.form.getOptionData(),
                    action: {
                        method: 'approve',
                        button_code: code,
                        params: {}
                    }
                }

                await defHttp.post({
                    url: `tickets/${ticketInfo.id}/handle/`,
                    params
                })

                this.visible = false
                Message.success($t('审批成功'))

                if (typeof this.refreshTicket === 'function') {
                    await this.refreshTicket('approve', true)
                } else {
                    window.location.reload()
                }
            } catch (error) {
                Message.error(error?.message || $t('审批失败'))
            } finally {
                this.loading = false
            }
        }
    }
}
</script>

<style lang="scss" scoped>
.approve-tip {
    margin-bottom: 4px;
}
</style>
